<task type="auto">
  <name>Task 1: OpenRent Scraper Implementation</name>
  <files>src/libs/scrapers/openrent.ts</files>
  <action>
    Implement a scraper for OpenRent using `camoufox-js` and `playwright-core`.
    - Function `scrapeOpenRent(url: string)`:
      - Launch Camoufox.
      - Navigate to URL.
      - Extract: Price, Bedrooms, Property Type, Title, Lat/Lng (if available in scripts/meta), and unique ID.
      - Return a `Property` object.
    Handle basic navigation errors and timeouts.
  </action>
  <verify>
    <automated>npx ts-node src/libs/scrapers/openrent.ts --test-url="https://www.openrent.co.uk/property-to-rent/london/2-bed-flat/12345"</automated>
  </verify>
  <done>Scraper can extract data from a sample OpenRent property page.</done>
</task>

<task type="auto">
  <name>Task 2: BullMQ Scraper Worker</name>
  <files>src/workers/scraper-worker.ts</files>
  <action>
    Implement a BullMQ worker that:
    - Listens to 'scrapers' queue.
    - Receives property URLs.
    - Executes `scrapeOpenRent`.
    - Upserts the result into the PostgreSQL `properties` table using Drizzle.
    Use `st_point(lng, lat)` for the location column.
  </action>
  <verify>
    <automated>node --test src/workers/scraper-worker.test.ts</automated>
  </verify>
  <done>Worker processes jobs and updates database correctly.</done>
</task>

<task type="auto" tdd="true">
  <name>Task 3: Property API Endpoints</name>
  <files>src/apps/api/server.ts, src/apps/api/routes/properties.ts</files>
  <behavior>
    - GET /properties:
      - Accepts `bbox` (minLng, minLat, maxLng, maxLat).
      - Returns GeoJSON FeatureCollection of properties within bounds.
      - Use `ST_MakeEnvelope` and `ST_Within` in Drizzle.
    - POST /properties/scrape:
      - Accepts `{url}`.
      - Adds a job to the 'scrapers' queue.
  </behavior>
